def expand(token: str) -> list[str]:
    """Brace expansion like the shell's: a_{x,y}.py -> a_x.py, a_y.py."""

"""Gate list comparison (M28 T17): the documented gate against .codex/hooks/stop.py and frontend/package.json.

Run from the repository root (cwd). Standard library only; reads files (stop.py is parsed with ast, never imported),
prints a Markdown section to stdout; exit 0 only when every comparison matches.
"""
import ast
import json
import re
import sys
from pathlib import Path

values = {}
for node in stop.body:
    if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
        values[node.targets[0].id] = node.value
script_tests = [elt.value for elt in values["SCRIPT_TESTS"].elts]
pytest_groups = len(values["PYTEST_GROUPS"].elts)
checks = 1 + pytest_groups + 1          # stop.py's CHECKS expression: ruff + PYTEST_GROUPS + REST_GROUP
globbed = sorted(str(p) for p in Path("scripts").glob("test_*.py"))
scripts = json.loads(Path("frontend/package.json").read_text(encoding="utf-8"))["scripts"]
harnesses = sorted(str(p) for p in Path("frontend/tests/offscreen").glob("*.cjs"))


def block(path: str, text: str) -> str:
    match = re.search(r"## 검증 (게이트|명령)\n.*?```bash\n(.*?)```", text, re.S)
    if not match:
        raise SystemExit(f"no gate block in {path}")
    return match.group(2)


rows, ok = [], True
for path in ("docs/standards.md", "docs/operations.md"):
    code = block(path, Path(path).read_text(encoding="utf-8"))
    pytest_line = next(line for line in code.splitlines() if line.startswith(".venv/bin/python -m pytest scripts/"))
    listed = [tok for tok in pytest_line.split() if tok.startswith("scripts/test_")]
    node = [line.split()[-1] for line in code.splitlines() if line.startswith("node --check ")]

    same = listed == script_tests
    missing_files = [n for n in node if not Path(n).is_file()]
    endtoend = [n for n in node if "/endtoend-" in n]
    npm_ok = all(name in scripts for name in npm)
    ok &= same and not missing_files and len(endtoend) == 3 and npm_ok
    rows.append((path, len(listed), same, len(node), missing_files, endtoend, npm, npm_ok))

agents = Path("scripts/AGENTS.md").read_text(encoding="utf-8")
